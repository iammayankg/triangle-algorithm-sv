\begin{abstract}
The Triangle Algorithm (TA) solves hard-margin support vector machine
training as a convex-hull distance problem with a certified duality gap.
We extend the enhanced TA of Gupta and Kalantari to the full SVM family
--- $L_2$ and $L_1$ (hinge/$\nu$-SVM) soft margins and kernels --- through
exact reductions that reuse the algorithm's dot-product caches, and we
introduce three algorithmic improvements: \emph{block weight transfers}
that amortise each $O(n)$ scan over $k$ MDM-style transfers via one exact
line search along an aggregated direction; \emph{gap-certified safe
screening} that provably removes non-support points while preserving all
certificates; and a \emph{guarded step rule} with an away-step fallback.
Our main theoretical result is global linear convergence of the guarded
block algorithm with explicit constants and no combinatorial swap-step
bounds, obtained by analysing the method on the Minkowski-difference
polytope and inheriting the pyramidal-width machinery of Lacoste-Julien
and Jaggi. Empirically, the optimised solver is the fastest method to a
certified solution on six of seven benchmark classes against LIBSVM,
LIBLINEAR, and SMO baselines, and every inequality in the analysis is
verified numerically at machine precision.
\end{abstract}

\section{Background and setting}
Let $V=\{v_1,\dots,v_n\}$, $W=\{w_1,\dots,w_m\}\subset\mathbb{R}^d$ and
\begin{equation}
  \min_{p\in\conv(V),\,q\in\conv(W)} f(p,q)=\tfrac12\norm{p-q}^2 .
  \label{eq:distance}
\end{equation}
With $z=p-q$ and the Minkowski-difference polytope
$Z=\conv(V)-\conv(W)=\conv\{v_i-w_j\}$, \eqref{eq:distance} is the
minimum-norm-point problem over $Z$, whose objective is $1$-strongly
convex and $1$-smooth. The Triangle Algorithm~\cite{kalantari,gupta}
maintains an iterate pair with convex weights $(\alpha,\beta)$ and a
certified interval $[\mathrm{LB},\mathrm{UB}]$ from parallel supporting
hyperplanes; it stops when
$\mathrm{UB}-\mathrm{LB}\le\varepsilon\,\mathrm{UB}$.

$L_2$ soft margin: $K\mapsto K+I/C$ via slack coordinates
$\tilde x_i=[x_i;e_i/\sqrt{C}]$, never materialised (sparse diagonal

$R(V,\mu)=\{\sum a_i v_i:\ \sum a_i=1,\ 0\le a_i\le\mu\}$,
$\mu=2/(\nu\ell)$; the extreme-point oracle becomes a capped top-$k$
blend. Kernels: all quantities are inner products; iterates live in

\section{The guarded block-transfer algorithm}
\begin{algorithm}[h]
\caption{Guarded block transfer with away fallback (one side, one iteration)}
\begin{algorithmic}[1]
\State scores $s_i\gets\ip{g}{v_i}$;\quad top-$k$ receivers, worst-$k$
active donors, paired best-with-worst; pair 1 is the MDM pair $(r^*,u^*)$
\State per-pair steps $\gamma_j\gets\min(\mathrm{gap}_j/s_j,\,c_j)$
\If{pair 1 uncapped}
  \State take the better of the block step (exact line search along
  $d=\sum_j\gamma_j d_j$) and the single MDM step \Comment{guard}
\Else
  \If{away step on $u^*$ is boundary-clipped}
     \State take it \Comment{drop step: $\alpha_{u^*}\to 0$}
  \Else
     \State take the best of \{block, MDM, away($u^*$), toward($r^*$)\}
  \EndIf
\EndIf
\end{algorithmic}
\end{algorithm}

\begin{lemma}[Gap separability]
With the product representation $\lambda_{ij}=\alpha_i\beta_j$ over $Z$,
$g^{\mathrm{PW}}=g^{\mathrm{PW}}_V+g^{\mathrm{PW}}_W$, and likewise for
the Frank--Wolfe and away gaps; side transfers are pairwise directions of
$Z$.
\end{lemma}

\begin{lemma}[Block gain]
With $S=\ip{g}{d}$, $D=\norm{d}^2$: $S\ge\sum_j\gamma_j^2 s_j$,
$D\le kS$, and the realised block gain satisfies
$\Delta_B\ge S/(2k)\ge\Delta_{\mathrm{MDM}}/(2k)$.
\end{lemma}

\begin{theorem}[Global linear convergence]
For any block size $k\ge1$, the guarded block-transfer Triangle Algorithm
with away fallback satisfies, with $\delta=\PW(Z)$, $M=\operatorname{diam}(Z)$,
$\rho=c\,\delta^2/M^4$ for an explicit absolute constant $c$:
every non-drop iteration contracts, $h_{t+1}\le(1-\rho)h_t$; drop
iterations number at most $k\,T_{\mathrm{good}}+s_0$; hence
$h_T\le h_0(1-\rho)^{(T-s_0)/(k+2)}$.
\end{theorem}

\begin{proposition}[Near-orthogonal speedup]
If $|\ip{d_i}{d_j}|\le\eta\sqrt{s_i s_j}$ for $i\ne j$, the interior
block gain is at least $(\sum_j\Delta_j)/(1+\eta(k-1))$.
\end{proposition}

\begin{theorem}[Screening activation]
Let $\delta^*>0$, let $\tau$ be the minimal nonzero optimal-face score
margin, $D$ the larger class diameter and $R=\max_{z\in Z}\norm{z}$. The
certified gap equals the Frank--Wolfe gap over $Z$ divided by
$\mathrm{UB}$, and once
$h_t \le \tau^4\delta^{*2}/(2592\,R^4 D^4)$
every zero-weight point outside the optimal faces is screened at the
next round, permanently; under the linear rate this holds for all
$t \ge T^* = s_0 + \tfrac{k+2}{\rho}\,
\log\!\bigl(2592\,R^4D^4 h_0/(\tau^4\delta^{*2})\bigr)$,
after which every scan costs $O(|F|+|\mathrm{supp}|)$ instead of
$O(n+m)$. The $\tau$-dependence is information-theoretically necessary.
\end{theorem}

\begin{theorem}[Safe screening]
With $r=\sqrt{\mathrm{UB}^2-\mathrm{LB}^2}$, any zero-weight point with
$(\ip{h}{v_i}-\ip{h}{v_{\min}})>r\norm{v_i-v_{\min}}$ is outside the
optimal support; removing all such points preserves the optimum and the
validity of all subsequent lower bounds.
\end{theorem}
